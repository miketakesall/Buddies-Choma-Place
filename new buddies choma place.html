🛠 Developer: Real Daraja API Setup
How STK Push Works

The frontend cannot call Safaricom Daraja directly (CORS + security). You need a backend that holds your API credentials. Here's a complete setup:

1. Get Credentials from Safaricom
  • Register at developer.safaricom.co.ke
  • Create an app → get Consumer Key & Consumer Secret
  • Your Business Short Code = 8462960 (your Till Number)
  • Get your Lipa Na Mpesa Online Passkey from the portal
  • Set a Callback URL (public HTTPS endpoint)
2. Node.js/Express Backend (server.js)
// npm install express axios cors dotenv
// ─── .env file ─────────────────────────────────
// MPESA_CONSUMER_KEY=your_key_here
// MPESA_CONSUMER_SECRET=your_secret_here
// MPESA_PASSKEY=your_passkey_here
// MPESA_SHORTCODE=8462960
// MPESA_CALLBACK_URL=https://yourdomain.com/api/mpesa/callback
// ────────────────────────────────────────────────

require('dotenv').config();
const express = require('express');
const axios   = require('axios');
const cors    = require('cors');
const app     = express();
app.use(cors({ origin: 'https://yoursite.com' })); // restrict to your domain
app.use(express.json());

const BASE = 'https://sandbox.safaricom.co.ke'; // change to api.safaricom.co.ke for LIVE

// ── Step 1: OAuth Token ──────────────────────────────────────
async function getAccessToken() {
  const auth = Buffer.from(
    `${process.env.MPESA_CONSUMER_KEY}:${process.env.MPESA_CONSUMER_SECRET}`
  ).toString('base64');
  const res = await axios.get(`${BASE}/oauth/v1/generate?grant_type=client_credentials`,
    { headers: { Authorization: `Basic ${auth}` } });
  return res.data.access_token;
}

// ── Step 2: STK Push Endpoint ────────────────────────────────
app.post('/api/mpesa/stkpush', async (req, res) => {
  try {
    const { phoneNumber, amount } = req.body;
    // Validate inputs server-side
    if (!/^2547\d{8}$/.test(phoneNumber)) return res.status(400).json({ error: 'Invalid phone' });
    if (!amount || amount < 1)           return res.status(400).json({ error: 'Invalid amount' });

    const token     = await getAccessToken();
    const timestamp = new Date().toISOString().replace(/[^0-9]/g, '').slice(0, 14);
    const password  = Buffer.from(
      `${process.env.MPESA_SHORTCODE}${process.env.MPESA_PASSKEY}${timestamp}`
    ).toString('base64');

    const payload = {
      BusinessShortCode: process.env.MPESA_SHORTCODE,  // 8462960
      Password:          password,
      Timestamp:         timestamp,
      TransactionType:   'CustomerBuyGoodsOnline',     // Buy Goods (Till)
      Amount:            Math.ceil(amount),             // No decimals
      PartyA:            phoneNumber,                   // Customer phone
      PartyB:            process.env.MPESA_SHORTCODE,  // Till number
      PhoneNumber:       phoneNumber,
      CallBackURL:       process.env.MPESA_CALLBACK_URL,
      AccountReference:  'BuddiesChoma',
      TransactionDesc:   'Food Order - Buddies Choma Place'
    };

    const mpesaRes = await axios.post(
      `${BASE}/mpesa/stkpush/v1/processrequest`, payload,
      { headers: { Authorization: `Bearer ${token}` } }
    );
    res.json(mpesaRes.data);
  } catch (err) {
    console.error('STK Push error:', err.response?.data || err.message);
    res.status(500).json({ error: 'Payment initiation failed' });
  }
});

// ── Step 3: Callback (Safaricom calls this after payment) ─────
app.post('/api/mpesa/callback', (req, res) => {
  const callback = req.body?.Body?.stkCallback;
  if (callback?.ResultCode === 0) {
    const meta    = callback.CallbackMetadata.Item;
    const receipt = meta.find(i => i.Name === 'MpesaReceiptNumber')?.Value;
    const amount  = meta.find(i => i.Name === 'Amount')?.Value;
    const phone   = meta.find(i => i.Name === 'PhoneNumber')?.Value;
    console.log(`✅ Payment OK — Receipt: ${receipt}, KSH ${amount}, Phone: ${phone}`);
    // → Update your database / send confirmation here
  } else {
    console.log(`❌ Payment failed: ${callback?.ResultDesc}`);
  }
  res.json({ ResultCode: 0, ResultDesc: 'Accepted' });
});

// ── Step 4: Status check (optional polling) ───────────────────
app.get('/api/mpesa/status/:checkoutId', async (req, res) => {
  const token     = await getAccessToken();
  const timestamp = new Date().toISOString().replace(/[^0-9]/g,'').slice(0,14);
  const password  = Buffer.from(
    `${process.env.MPESA_SHORTCODE}${process.env.MPESA_PASSKEY}${timestamp}`
  ).toString('base64');
  const result = await axios.post(`${BASE}/mpesa/stkpushquery/v1/query`, {
    BusinessShortCode: process.env.MPESA_SHORTCODE,
    Password: password, Timestamp: timestamp,
    CheckoutRequestID: req.params.checkoutId
  }, { headers: { Authorization: `Bearer ${token}` } });
  res.json(result.data);
});

app.listen(3000, () => console.log('🔥 Server running on port 3000'));
3. Frontend Real API Call
// Replace simulateSTKPush() call in initPayment() with:
async function realSTKPush(phoneNumber, amount) {
  const res = await fetch('/api/mpesa/stkpush', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ phoneNumber, amount })
    // phoneNumber must be 2547XXXXXXXX format
    // amount is the cart total in KSH (integer)
  });
  const data = await res.json();
  // data.ResponseCode === '0' means STK Push sent successfully
  return data;
}
4. Security Reminders
  • Never put Consumer Key/Secret in frontend JS
  • Always validate phone + amount on the backend
  • Use HTTPS for callback URL (Safaricom requirement)
  • Restrict CORS to your domain only
  • For production: change sandbox URL to api.safaricom.co.ke
