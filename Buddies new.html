What You Need (Safaricom Daraja API)

To enable real M-Pesa STK Push payments, you need a backend server. The frontend cannot call Daraja directly due to CORS and security. Follow these steps:

  • Register at developer.safaricom.co.ke and create an app
  • Get your Consumer Key and Consumer Secret
  • Get your Business Shortcode (Till Number: 8462960)
  • Get your Lipa Na Mpesa Online Passkey from the portal
  • Set up a callback URL (publicly accessible HTTPS endpoint)
Backend Endpoint (Node.js / Express)
// npm install express axios dotenv // POST /api/mpesa/stkpush const express = require('express'); const axios = require('axios'); const app = express(); app.use(express.json()); const CONSUMER_KEY = process.env.MPESA_CONSUMER_KEY; const CONSUMER_SECRET = process.env.MPESA_CONSUMER_SECRET; const SHORTCODE = '8462960'; const PASSKEY = process.env.MPESA_PASSKEY; const CALLBACK_URL = 'https://yourdomain.com/api/mpesa/callback'; // Step 1: Get OAuth access token async function getAccessToken() { const auth = Buffer.from(`${CONSUMER_KEY}:${CONSUMER_SECRET}`).toString('base64'); const res = await axios.get( 'https://sandbox.safaricom.co.ke/oauth/v1/generate?grant_type=client_credentials', { headers: { Authorization: `Basic ${auth}` } } ); return res.data.access_token; } // Step 2: STK Push app.post('/api/mpesa/stkpush', async (req, res) => { const { phoneNumber, amount, tillNumber } = req.body; const token = await getAccessToken(); const timestamp = new Date().toISOString().replace(/[^0-9]/g,'').slice(0,14); const password = Buffer.from(`${SHORTCODE}${PASSKEY}${timestamp}`).toString('base64'); const payload = { BusinessShortCode: SHORTCODE, Password: password, Timestamp: timestamp, TransactionType: 'CustomerBuyGoodsOnline', Amount: Math.ceil(amount), PartyA: phoneNumber, PartyB: SHORTCODE, PhoneNumber: phoneNumber, CallBackURL: CALLBACK_URL, AccountReference: 'BuddiesChoma', TransactionDesc: 'Food Order - Buddies Choma Place' }; const mpesaRes = await axios.post( 'https://sandbox.safaricom.co.ke/mpesa/stkpush/v1/processrequest', payload, { headers: { Authorization: `Bearer ${token}` } } ); res.json(mpesaRes.data); }); // Step 3: Callback (Safaricom calls this after payment) app.post('/api/mpesa/callback', (req, res) => { const result = req.body.Body.stkCallback; if (result.ResultCode === 0) { // Payment successful — update your DB here const code = result.CallbackMetadata.Item.find(i => i.Name === 'MpesaReceiptNumber').Value; console.log('Payment success. Receipt:', code); } res.json({ ResultCode: 0, ResultDesc: 'Accepted' }); }); app.listen(3000);

// Replace simulateStkPush() with this for live payments: async function realStkPush(phoneNumber, amount) { const res = await fetch('/api/mpesa/stkpush', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ phoneNumber, // e.g. "254712345678" amount, // e.g. 1500 tillNumber: '8462960' }) }); const data = await res.json(); if (data.ResponseCode === '0') { // STK Push sent successfully — show PIN prompt to user return { success: true, checkoutId: data.CheckoutRequestID }; } else { return { success: false, message: data.ResponseDescription }; } } // Poll payment status (or use webhook/callback) async function checkPaymentStatus(checkoutId) { const res = await fetch(`/api/mpesa/status/${checkoutId}`); return await res.json(); }
