The Craft

FROM COAL TO PLATE

01
FRESH SOURCING, EVERY MORNING
We source our beef, goat, and offal from trusted local butchers every single morning. Nothing sits in a fridge overnight — freshness is non-negotiable at Buddies.
02
HAND-RUBBED & SEASONED
Each cut gets a hand-rub of sea salt, a whisper of spice, and nothing more. We believe good meat needs no disguising — just fire and time.
03
SLOW CHARCOAL FIRE
We use real hardwood charcoal, not briquettes. The grill is maintained at the right temperature for slow, even cooking that seals in all the juices.
04
CARVED & SERVED IMMEDIATELY
When your order is ready, we hand-chop it on a wooden board and plate it with fresh kachumbari and your sides. No waiting under heat lamps — straight from the fire to the table.

Express Server (server.js)
const express = require('express');
const axios   = require('axios');
const cors    = require('cors');
require('dotenv').config();
const app = express();
app.use(cors({ origin: '*' }));
app.use(express.json());

async function getToken() {
  const creds = Buffer.from(
    `${process.env.MPESA_CONSUMER_KEY}:${process.env.MPESA_CONSUMER_SECRET}`
  ).toString('base64');
  const r = await axios.get(
    'https://api.safaricom.co.ke/oauth/v1/generate?grant_type=client_credentials',
    { headers: { Authorization: `Basic ${creds}` } }
  );
  return r.data.access_token;
}

app.post('/api/mpesa/stkpush', async (req, res) => {
  const { phoneNumber, amount } = req.body;
  const token = await getToken();
  const ts = new Date().toISOString().replace(/[^0-9]/g,'').slice(0,14);
  const pass = Buffer.from(
    `${process.env.MPESA_SHORTCODE}${process.env.MPESA_PASSKEY}${ts}`
  ).toString('base64');
  const r = await axios.post(
    'https://api.safaricom.co.ke/mpesa/stkpush/v1/processrequest',
    {
      BusinessShortCode: process.env.MPESA_SHORTCODE,
      Password: pass, Timestamp: ts,
      TransactionType: 'CustomerBuyGoodsOnline',
      Amount: Math.ceil(amount),
      PartyA: phoneNumber, PartyB: process.env.MPESA_SHORTCODE,
      PhoneNumber: phoneNumber,
      CallBackURL: process.env.MPESA_CALLBACK_URL,
      AccountReference: 'BuddiesChoma',
      TransactionDesc: 'Food Order Payment'
    },
    { headers: { Authorization: `Bearer ${token}` } }
  );
  res.json(r.data);
});

app.post('/api/mpesa/callback', (req, res) => {
  const cb = req.body.Body.stkCallback;
  if (cb.ResultCode === 0) {
    const code = cb.CallbackMetadata.Item
      .find(i => i.Name === 'MpesaReceiptNumber').Value;
    console.log('SUCCESS:', code);
  }
  res.json({ ResultCode: 0, ResultDesc: 'Accepted' });
});

app.listen(process.env.PORT || 3001);
